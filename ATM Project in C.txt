
#include <stdio.h>

int main() {
    int pin, choice;
    double balance = 5000.00, amount;
    int correctPin = 1234;

    printf("================================\n");
    printf("       WELCOME TO ATM\n");
    printf("================================\n");

    printf("Enter your 4-digit PIN: ");
    scanf("%d", &pin);

    if (pin != correctPin) {
        printf("Incorrect PIN! Access denied.\n");
        return 0;
    }

    printf("\nLogin successful!\n");

    do {
        printf("\n========== ATM MENU ==========\n");
        printf("1. Check Balance\n");
        printf("2. Deposit Money\n");
        printf("3. Withdraw Money\n");
        printf("4. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch (choice) {
            case 1:
                printf("Current Balance: Rs. %.2lf\n",
                       balance);
                break;

            case 2:
                printf("Enter deposit amount: Rs. ");
                scanf("%lf", &amount);

                if (amount > 0) {
                    balance += amount;
                    printf("Deposit successful!\n");
                    printf("New Balance: Rs. %.2lf\n",
                           balance);
                } else {
                    printf("Invalid deposit amount!\n");
                }
                break;

            case 3:
                printf("Enter withdrawal amount: Rs. ");
                scanf("%lf", &amount);

                if (amount <= 0) {
                    printf("Invalid withdrawal amount!\n");
                } else if (amount > balance) {
                    printf("Insufficient balance!\n");
                } else {
                    balance -= amount;
                    printf("Please collect your money.\n");
                    printf("Remaining Balance: Rs. %.2lf\n",
                           balance);
                }
                break;

            case 4:
                printf("Thank you for using our ATM!\n");
                break;

            default:
                printf("Invalid choice! Try again.\n");
        }

    } while (choice != 4);

    return 0;
}
